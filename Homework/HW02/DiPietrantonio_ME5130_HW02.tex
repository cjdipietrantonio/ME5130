\textbf{18B.19, Problem Statement:}
\textit{Oxygen uptake by a bacterial aggregate.} Under suitable circumstances the rate of oxygen metabolism by bacterial cells is very nearly zero order with respect to oxygen concentration. We examine such a case here and focus our attention on a spherical aggregate of cells, which has a radius $R$. We wish to determine the total rate of oxygen uptake by the aggregate as a function of aggregate size, oxygen mass concentration $\rho_0$ at the aggregate surface, the metabolic activity of the cells, and the diffusional behavior of the oxygen. For simplicity we consider the aggregate to be homogeneous. We then approximate the metabolic rate by an effective volumetric reaction rate $r_{\mathrm{O}_2} = -k_0'''$ and the diffusional behavior by Fick's law, with an effective pseudobinary diffusivity $\mathcal{D}_{\mathrm{O}_2 m}$. Because the solubility of oxygen is very low in this system, both convective oxygen transport and transient effects may be neglected.

\begin{enumerate}
    \item[(a)] Show by means of a shell mass balance that the pseudo-steady-state oxygen concentration profile is described by the differential equation

    \begin{equation}
        \frac{1}{\xi^2} \frac{d}{d\xi} \left( \xi^2 \frac{d\chi}{d\xi} \right) = N
        \tag{18B.19--1}
    \end{equation}
    where $\chi = \rho_{\mathrm{O}_2} / \rho_0$, $\xi = r/R$, and $N = k_0''' R^2 / \rho_0 \mathcal{D}_{\mathrm{O}_2 m}$.

    \item[(b)] There may be an oxygen-free core in the aggregate, if $N$ is sufficiently large, such that $\chi = 0$ for $\xi < \xi_0$. Write sufficient boundary conditions to integrate Eq.~18B.19--1 for this situation. To do this, it must be recognized that both $\chi$ and $d\chi / d\xi$ are zero at $\xi = \xi_0$. What is the physical significance of this last statement?

    \item[(c)] Perform the integration of Eq.~18B.19--1 and show how $\xi_0$ may be determined.

    \item[(d)] Sketch the total oxygen uptake rate and $\xi_0$ as functions of $N$, and discuss the possibility that no oxygen-free core exists.
\end{enumerate}

\textit{Answer:} (c)

\begin{equation}
    \chi = 1 - \frac{N}{6} \left( 1 - \xi^2 \right) + \frac{N}{3} \xi_0^3 \left( \frac{1}{\xi} - 1 \right)
\end{equation}
for $\xi \geq \xi_0 \geq 0$, where $\xi_0$ is determined as a function of $N$ from

\begin{equation}
    \xi_0^3 - \frac{3}{2} \xi_0^2 + \left( \frac{1}{2} - \frac{3}{N} \right) = 0
\end{equation}

\vspace{6pt}

\textbf{18B.19, Solution:}

To solve this problem, we can consider the oxygen going into and out of a thin spherical shell between radius $r$ and $r + \Delta r$. The rate of oxygen going in through the outer surface is:

\begin{equation}
    - n_{O_{2}, r} \bigg|_{r + \Delta r} \cdot 4 \pi (r + \Delta r)^2
\end{equation}

The rate of oxygen going out of the inner surface is:

\begin{equation}
    - n_{O_{2}, r} \bigg|_{r} \cdot 4 \pi r^2
\end{equation}

and the rate of oxygen being produced within the shell is:

\begin{equation}
    -k_0''' \cdot 4 \pi r^2 \Delta r
\end{equation}

Therefore, the shell balance can be constructed as:

\begin{equation}
    n_{O_{2}, r} \bigg|_{r} \cdot 4 \pi r^2 - n_{O_{2}, r} \bigg|_{r + \Delta r} \cdot 4 \pi (r + \Delta r)^2 - k_0''' \cdot 4 \pi r^2 \Delta r = 0
\end{equation}

Dividing by $4 \pi \Delta r$:

\begin{equation}
\begin{aligned}
    \frac{\left( r^2 n_{O_{2}, r} \right) \bigg|_{r} - \left( r^2 n_{O_{2}, r} \right) \bigg|_{r + \Delta r}}{\Delta r} - k_0''' r^2 &= 0 \\[6pt]
    \frac{\left( r^2 n_{O_{2}, r} \right) \bigg|_{r + \Delta r} - \left( r^2 n_{O_{2}, r} \right) \bigg|_{r}}{\Delta r} &= -k_0''' r^2 
\end{aligned}
\end{equation}

We can now consider the definition of a derivative:

\begin{equation}
    \frac{df}{dr} = \lim_{\Delta r \to 0} \frac{f(r + \Delta r) - f(r)}{\Delta r}
\end{equation}

Therefore, if we take $\lim_{\Delta r \to 0}$, we have:

\begin{equation}
    \frac{d}{dr} \left( r^2 n_{O_{2}, r} \right) = -k_0''' r^2
\end{equation}

From Fick's law, we have:

\begin{equation}
    n_{O_{2,}, r} = - \mathcal{D} \frac{d\rho_{\mathrm{O}_2}}{dr}
\end{equation}

and can write:

\begin{equation}
\begin{aligned}
    \frac{d}{dr} \left( r^2 \left( - \mathcal{D} \frac{d\rho_{\mathrm{O}_2}}{dr}\right) \right) &= -k_0''' r^2 \\
    \mathcal{D} \frac{d}{dr} \left( r^2 \frac{d\rho_{\mathrm{O}_2}}{dr}\right)                  &= k_0''' r^2 \\
\end{aligned}
\end{equation}

Since $\xi = r/R$:

\begin{equation}
    r = R \xi
\end{equation}

\begin{equation}
    dr = R d \xi
\end{equation}

Therefore:

\begin{equation}
    \frac{d}{dr} = \frac{1}{R} \frac{d}{d \xi}
\end{equation}

and we can write:

\begin{equation}
\begin{aligned}
    \mathcal{D} \frac{1}{R} \frac{d}{d \xi} \left( R^2 \xi^2 \frac{1}{R} \frac{d\rho_{\mathrm{O}_2}}{d \xi}\right) &= k_0''' R^2 \xi^2 \\
    \mathcal{D} \frac{d}{d \xi} \left( \xi^2 \frac{d\rho_{\mathrm{O}_2}}{d \xi}\right) &= k_0''' R^2 \xi^2 \\
\end{aligned}
\end{equation}

and since $\chi = \rho_{\mathrm{O}_2} / \rho_0$:

\begin{equation}
    \rho_{\mathrm{O}_2} = \chi \rho_0
\end{equation}

\begin{equation}
    \frac{d \rho_{\mathrm{O}_2}}{d \xi} = \rho_0 \frac{d \chi}{d \xi}
\end{equation}

therefore:

\begin{equation}
\begin{aligned}
    \mathcal{D} \frac{d}{d \xi} \left( \xi^2 \rho_0 \frac{d \chi}{d \xi}\right) &= k_0''' R^2 \xi^2 \\
    \mathcal{D} \rho_0 \frac{d}{d \xi} \left( \xi^2 \frac{d \chi}{d \xi}\right) &= k_0''' R^2 \xi^2 \\
    \frac{1}{\xi^2} \frac{d}{d \xi} \left( \xi^2 \frac{d \chi}{d \xi}\right) &= \frac{k_0''' R^2}{\mathcal{D} \rho_0 }
\end{aligned}
\end{equation}

and since:

\begin{equation}
    N = \frac{k_0''' R^2}{\rho_0 \mathcal{D}_{\mathrm{O}_2 m}}
\end{equation}

We have:

\begin{equation}
\boxed{
    \frac{1}{\xi^2} \frac{d}{d \xi} \left( \xi^2 \frac{d \chi}{d \xi}\right) = N
}
\end{equation}

If there is an oxygen-free core in the aggregate, the cells use oxygen faster than diffusion can bring it in, so the oxygen must run out before reaching the center such that $\chi = 0$ for $\xi < \xi_0$. Therefore, the sphere has two regions: the outer shell where oxygen is present:

\begin{equation}
    \xi_0 < \xi \leq 1 
\end{equation}

And the core, where no oxygen is present:

\begin{equation}
    0 \leq \xi \leq \xi_0
\end{equation}

At the outer surface, $\rho_{O_2} = \rho_0$, therefore:

\begin{equation}
\boxed{
    \chi = 1 \quad \mathrm{at} \quad \xi = 1
}
\end{equation}

At the edge of the core:

\begin{equation}
\boxed{
    \chi = 0 \quad \mathrm{at} \quad \xi = \xi_0
}
\end{equation}

\begin{equation}
\boxed{
    \frac{d \chi}{d \xi} = 0 \quad \mathrm{at} \quad \xi = \xi_0
}
\end{equation}

The physical significance of this is that when $\chi = 0$ at $\xi_0$, the oxygen runs out exactly at the edge of the core. Furthermore, when $\frac{d \chi}{d \xi} = 0$ at $\xi_0$, no oxygen is flowing into the core. We know this because from part a we have:

\begin{equation}
    n_{O_{2,}, r} = - \mathcal{D} \frac{\rho_0}{R}\frac{d \chi}{d \xi}
\end{equation}

From this we can see that the mass flux is proportional to the slope of the normalized mass concentration with respect to the normalized radius. Therefore, if this slope is zero at $\xi = \xi_0$, then so is the mass flux. We can now integrate Eq.~18B.19--1:

\begin{equation}
\begin{aligned}
    \frac{1}{\xi^2} \frac{d}{d\xi} \left( \xi^2 \frac{d\chi}{d\xi} \right) &= N \\
    \frac{d}{d\xi} \left( \xi^2 \frac{d\chi}{d\xi} \right) &= N\xi^2 \\
    \int \frac{d}{d\xi} \left( \xi^2 \frac{d\chi}{d\xi} \right) d \xi &=  \int N\xi^2 d \xi\\
    \xi^2 \frac{d\chi}{d\xi} &= \frac{N\xi^3}{3} + C_1 \\
    \frac{d\chi}{d\xi} &= \frac{N\xi}{3} + \frac{C_1}{\xi^2} \\
\end{aligned}
\end{equation}

Now, we know that at $\xi = \xi_0$, $\frac{d \chi}{d \xi} = 0$:

\begin{equation}
\begin{aligned}
    0   &= \frac{N}{3} \xi_0 + \frac{C_1}{\xi^2_0} \\
    0   &= \frac{N}{3} \xi^3_0 + C_1 \\
    C_1 &= - \frac{N}{3} \xi^3_0
\end{aligned}
\end{equation}

Therefore:

\begin{equation}
\begin{aligned}
    \frac{d\chi}{d\xi} &= \frac{N\xi}{3} - \frac{N}{3} \frac{\xi^3_0}{\xi^2} \\
    \frac{d\chi}{d\xi} &= \frac{N}{3} \left( \xi - \frac{\xi^3_0}{\xi^2} \right) \\
\end{aligned}
\end{equation}

We can now evaluate:

\begin{equation}
\begin{aligned}
    \int \frac{d\chi}{d\xi} d \xi &= \frac{N}{3} \int \left( \xi - \frac{\xi^3_0}{\xi^2} \right) d \xi \\
    \chi                          &= \frac{N}{3} \left( \frac{\xi^2}{2} + \frac{\xi^3_0}{\xi} \right) + C_2 \\
\end{aligned}
\end{equation}

We also know that at $\xi = \xi_0$, $\chi = 0$:

\begin{equation}
\begin{aligned}
    0   &= \frac{N}{3} \left( \frac{\xi^2_0}{2} + \frac{\xi^3_0}{\xi_0} \right) + C_2 \\
    0   &= \frac{N}{3} \left( \frac{\xi^2_0}{2} + \xi^2_0 \right) + C_2 \\
    0   &= \frac{N}{2} \xi^2_0 + C_2 \\
    C_2 &= -\frac{N}{2} \xi^2_0
\end{aligned}
\end{equation}

Thus:

\begin{equation}
\begin{aligned}
    \chi &= \frac{N}{3} \left( \frac{\xi^2}{2} + \frac{\xi_0^3}{\xi} \right) - \frac{N}{2} \xi^2_0 \\
        &= \frac{N}{3} \left( \frac{\xi^2}{2} + \frac{\xi_0^3}{\xi} - \frac{3}{2} \xi^2_0 \right) \\
\end{aligned}
\end{equation}

And finally, we can write:

\begin{equation}
\boxed{
\begin{aligned}
    \chi &= \frac{N}{3} \left( \frac{\xi^2}{2} + \frac{\xi_0^3}{\xi} - \frac{3}{2} \xi^2_0 \right), \qquad \xi_0 < \xi \leq 1\\
\end{aligned}
}
\end{equation}

and:

\begin{equation}
\boxed{
    \chi = 0, \qquad 0 \leq \xi \leq \xi_0
}
\end{equation}

To show how $\xi_0$ may be determined, we can recall that at $\xi = 1$, $\chi = 1$. Therefore:

\begin{equation}
\begin{aligned}
    1           &= \frac{N}{3} \left( \frac{1}{2} + \xi_0^3 - \frac{3}{2} \xi^2_0 \right) \\
    \frac{3}{N} &= \frac{1}{2} + \xi_0^3 - \frac{3}{2} \xi^2_0
\end{aligned}
\end{equation}

and thus:

\begin{equation}
\boxed{
    \xi_0^3 -\frac{3}{2} \xi_0^2 + \left( \frac{1}{2} - \frac{3}{N} \right) = 0
}
\end{equation}

Now, we know that the total oxygen uptake rate can be expressed as:

\begin{equation}
    W_{O_2} = -4\pi R^2 n_{O_2,r}\big|_{r=R}
\end{equation}

Which can be rewritten using Fick's law as:

\begin{equation}
    W_{O_2} = 4\pi R^2 \mathcal{D} \frac{d\rho_{O_2}}{dr} \big|_{r=R}
\end{equation}

Now, we have already shown that:

\begin{equation}
    \frac{d \rho_{O_2}}{d r} = \frac{\rho_0}{R} \frac{d \chi}{d \xi}
\end{equation}

So we have:

\begin{equation}
    W_{O_2} = 4\pi R \mathcal{D} \rho_0 \frac{d \chi}{d \xi} \big|_{\xi=1}
\end{equation}

Now we know that:

\begin{equation}
    \frac{d \chi}{d \xi} = \frac{N}{3} \left( \xi - \frac{\xi_0^3}{\xi^2} \right)
\end{equation}

Therefore:

\begin{equation}
    \frac{d \chi}{d \xi} \big|_{\xi=1} = \frac{N}{3} \left( 1 - \xi_0^3 \right)
\end{equation}

and so we can write:

\begin{equation}
        W_{O_2} = 4\pi R \mathcal{D} \rho_0 \frac{N}{3} \left( 1 - \xi_0^3 \right)
\end{equation}

and further, the non-dimensional uptake rate can be expressed as:

\begin{equation}
    W_{O_2}^* = \frac{W_{O_2}}{4\pi R \mathcal{D} \rho_0} = \frac{N}{3} \left( 1 - \xi_0^3 \right)
\end{equation}

Now, in the case where there is no core, the boundary condition at the center of the aggregate becomes $\frac{d \chi}{d \xi} = 0$ at $\xi = 0$ by symmetry. We can now re-evaluate:

\begin{equation}
    \frac{d \chi}{d \xi} = \frac{N}{3} \xi + \frac{C_1}{\xi^2}
\end{equation}

and since an infinite concentration gradient at the center of the aggregate is physically impossible, we know that $C_1 = 0$. Therefore:

\begin{equation}
    \frac{d \chi}{d \xi} = \frac{N}{3} \xi
\end{equation}

and:

\begin{equation}
    \frac{d \chi}{d \xi} \big|_{\xi=1} = \frac{N}{3}
\end{equation}

We should note that this could have also been obtained by substituting $\xi_0 = 0$ into our previous expression for the concentration gradient (in the case where there was a core). Nevertheless, we then have:

\begin{equation}
    W_{O_2, \mathrm{no \, core}}^* = \frac{W_{O_2, \mathrm{no \, core}}}{4\pi R \mathcal{D} \rho_0} = \frac{N}{3}
\end{equation}

We can now continue by writing:

\begin{equation}
\begin{aligned}
    \frac{d \chi}{d \xi} &= \frac{N}{3} \xi \\
    \int \frac{d \chi}{d \xi} d \xi &= \int\frac{N}{3} \xi d \xi \\
\end{aligned}
\end{equation}

Therefore:

\begin{equation}
    \chi = \frac{N}{6} \xi^2 + C_2
\end{equation}

And since $\chi = 1$ at $\xi = 1$:

\begin{equation}
\begin{aligned}
    1 &= \frac{N}{6} + C_2 \\
    C_2 &= 1 - \frac{N}{6}
\end{aligned}
\end{equation}

So

\begin{equation}
    \chi = \frac{N}{6} \xi^2 + 1 - \frac{N}{6}
\end{equation}

This means that at the center of the aggregate, $\xi = 0$, the normalized mass concentration is:

\begin{equation}
    \chi(0) = 1 - \frac{N}{6}
\end{equation}

Thus, for the case where there is no core, the condition above holds $(\chi(0) \geq 0)$ when $N \leq 6$. For $N > 6$, the normalized mass concentration at the center of the aggregate would become negative, which is physically impossible, so there must be a core. We can now plot the total oxygen uptake rate and $\xi_0$ as functions of $N$:

\begin{figure}[H]
    \centering
    \includegraphics[width=0.85\textwidth]{DiPietrantonio_ME5130_HW02_18B19.png}
\end{figure}

First, we can clearly see that as the rate of consumption of $O_2$ increases relative to the rate of diffusion ($N$ increases), the radius of the oxygen-free core increases. We can also observe that as $N$ increases, the non-dimensional uptake rate increases, while its rate of change decreases. This occurs because the rate of consumption increases with respect to diffusion, while oxygen is consumed in a smaller volume of the aggregate ($\xi_0$ increases).

\textbf{18C.2, Problem Statement:}
\textit{Diffusion and reaction in a partially impregnated catalyst.} Consider a catalytic sphere like that in \S 18.7, except that the active ingredient of the catalyst is present only in the annular region between $r = \kappa R$ and $r = R$:

\begin{align}
    &\text{In region I $\left( 0 < r < \kappa R \right)$,} & k_1'' a &= 0 \tag{18C.2--1} \\
    &\text{In region II $\left( \kappa R < r < R \right)$,} & k_1'' a &= \text{constant} > 0 \tag{18C.2--2}
\end{align}
Such a situation may arise when the active ingredient is put on the particles after pelleting, as is done for many commercial catalysts.

\begin{enumerate}
    \item[(a)] Integrate Eq.~18.7--6 separately for the active and inactive regions. Then apply the appropriate boundary conditions to evaluate the integration constants, and solve for the concentration profile in each region. Give qualitative sketches to illustrate the forms of the profiles.

    \item[(b)] Evaluate $W_{AR}$, the total molar rate of conversion of $A$ in a single particle.
\end{enumerate}

\vspace{6pt}

\textbf{18C.2, Solution:}

Equation 18.7-6 is given as:

\begin{equation}
    \mathcal{D}_A \frac{1}{r^2} \frac{d}{dr} \left(r^2 \frac{dc_A}{dr} \right) = k_1^{\prime\prime}a c_A
\end{equation}

In the region I, no reaction occurs, and we have:

\begin{equation}
\begin{aligned}
    \mathcal{D}_A \frac{1}{r^2} \frac{d}{dr} \left(r^2 \frac{dc_A}{dr} \right) &= 0 \\
    \frac{d}{dr} \left(r^2 \frac{dc_A}{dr} \right)                             &= 0 \\
    \int \frac{d}{dr} \left(r^2 \frac{dc_A}{dr} \right) \, dr                  &= \int 0 \, dr \\
    r^2 \frac{dc_A}{dr}                                                        &= C_1 \\
    \frac{dc_A}{dr}                                                            &= \frac{C_1}{r^2} \\
    \int \frac{dc_A}{dr} \, dr                                                 &= \int \frac{C_1}{r^2} \, dr\\ 
    c_A                                                                        &= - \frac{C_1}{r} + C_2
\end{aligned}
\end{equation}

Because $c_A$ must be finite at $r = 0$, $C_1 = 0$, thus:

\begin{equation}
    c_A = C_2
\end{equation}

This physically means that the concentration is uniform and its gradient is zero everywhere in the core (which makes sense because A does not react with anything in the core).

For region II we write:

\begin{equation}
    \frac{c_A}{c_{AR}} = \frac{f(r)}{r}
\end{equation}

where $c_{AR}$ is the concentration of $A$ at the surface ($r = R$). This reduces to equation 18.7-7:

\begin{equation}
    \frac{d^2 f}{d r^2} = \left( \frac{k''_1 a}{\mathcal{D}_A} \right) f
\end{equation}

If we let:

\begin{equation}
    b^2 = \frac{k''_1 a}{\mathcal{D}_A}
\end{equation}

we can write the solution to the differential equation above as:

\begin{equation}
    f = C_3 \cosh{br} + C_4 \sinh{br}
\end{equation}

We can then write:

\begin{equation}
    f = C_3 \cosh{b(r - \kappa R)} + C_4 \sinh{b(r - \kappa R)}
\end{equation}

and divide by $r$ to get:

\begin{equation}
    \frac{c_A}{c_{AR}} = \frac{1}{r} \left[ C_3 \cosh{b(r - \kappa R)} + C_4 \sinh{b(r - \kappa R)} \right]
\end{equation}

Now, we know that at $r = \kappa R$, $\frac{d c_A}{d r} = 0$, and since $\frac{c_A}{c_{AR}} = \frac{f(r)}{r}$:

\begin{equation}
\begin{aligned}
    \frac{1}{r} \frac{df}{dr} - \frac{f}{r^2}             &= 0 \\
    \kappa R \frac{df}{dr} \bigg|_{\kappa R} - f(\kappa R) &= 0 \\
    \kappa R \frac{df}{dr} \bigg|_{\kappa R} - C_3 &= 0
\end{aligned}
\end{equation}

We can write:

\begin{equation}
\begin{aligned}
    \frac{df}{dr}                   &= bC_3 \sinh{b(r - \kappa R)} + bC_4 \cosh{b(r - \kappa R)} \\
    \frac{df}{dr} \bigg|_{\kappa R} &= b C_4
\end{aligned}
\end{equation}

and then:

\begin{equation}
\begin{aligned}
    \kappa R b C_4 - C_3 &= 0 \\
    \kappa R b C_4       &= C_3 
\end{aligned}
\end{equation}

Where $\phi = bR = \sqrt{\frac{k''_1 a}{\mathcal{D}_A}} R$ is referred to as the ``Thiele modulus". So:

\begin{equation}
    C_3 = \kappa \phi C_4
\end{equation}

Next we know that at $r = R$, $\frac{c_A}{c_{AR}} = 1$:

\begin{equation}
\begin{aligned}
    \frac{c_A}{c_{AR}} &= \frac{1}{r} \left[ C_3 \cosh{b(r - \kappa R)} + C_4 \sinh{b(r - \kappa R)} \right] \\
    1                  &= \frac{1}{R} \left[ C_3 \cosh{b(R - \kappa R)} + C_4 \sinh{b(R - \kappa R)} \right] \\
    R                  &= C_3 \cosh{\phi(1 - \kappa)} + C_4 \sinh{\phi(1 - \kappa)} \\
    R                  &=  \kappa \phi C_4 \cosh{\phi(1 - \kappa)} + C_4 \sinh{\phi(1 - \kappa)} \\
    R                  &=  C_4 \left[\kappa \phi \cosh{\phi(1 - \kappa)} + \sinh{\phi(1 - \kappa)} \right]\\
\end{aligned}
\end{equation} 

Now, let $D = \kappa \phi \cosh{\phi(1 - \kappa)} + \sinh{\phi(1 - \kappa)}$:

\begin{equation}
    C_4 = \frac{R}{D}
\end{equation}

\begin{equation}
    C_3 = \frac{\kappa \phi R}{D}
\end{equation}

Therefore, for region II we have:

\begin{equation}
    \frac{c_A}{c_{AR}} = \frac{1}{r} \left[ \frac{\kappa \phi R}{D} \cosh{b(r - \kappa R)} + \frac{R}{D} \sinh{b(r - \kappa R)} \right]
\end{equation}

Which can be simplified to:

\begin{equation}
\boxed{
    \frac{c_A}{c_{AR}} = \frac{R}{rD} \left[ \kappa \phi \cosh{b(r - \kappa R)} + \sinh{b(r - \kappa R)} \right], \qquad \kappa R \leq r \leq R
}
\end{equation}

Returning to region I, we know that the concentration $c_A$ must be equal at the interface of the two regions $(r = \kappa R)$, so we have:

\begin{equation}
    \frac{c_A}{c_{AR}} = \frac{C_2}{c_{AR}}
\end{equation}

Therefore:

\begin{equation}
\begin{aligned}
    \frac{C_2}{c_{AR}} &= \frac{R}{\kappa R D} \left[\kappa \phi \cosh{0} +  \sinh{0}\right] \\
    \frac{C_2}{c_{AR}} &= \frac{\phi}{D}
\end{aligned}
\end{equation}

and finally:

\begin{equation}
\boxed{
    \frac{c_A}{c_{AR}} =\frac{\phi}{D}, \qquad 0 \leq r \leq \kappa R
}
\end{equation}

We can now give qualitative sketches to illustrate the forms of the profiles.

\begin{figure}[H]
    \centering
    \includegraphics[width=0.85\textwidth]{DiPietrantonio_ME5130_HW02_18C2.png}
\end{figure}

Now, to evaluate $W_{AR}$, the total molar rate of conversion of $A$ in a single particle, we can write:

\begin{equation}
\begin{aligned}
    W_{AR} &= 4 \pi R^2 N_{Ar} \bigg|_{r=R} \\
    W_{AR} &= - 4 \pi R^2 \mathcal{D}_A \frac{d c_A}{d r} \bigg|_{r=R}
\end{aligned}
\end{equation}

Since $\frac{c_A}{c_{AR}} = \frac{f(r)}{r}$:

\begin{equation}
\begin{aligned}
    \frac{d c_A}{d r} = \frac{c_{AR}}{r^2} \left( r \frac{df}{dr} - f(r) \right)
\end{aligned}
\end{equation}

We know that:

\begin{equation}
\begin{aligned}
    \frac{df}{dr} &= bC_3 \sinh{b(r - \kappa R)} + bC_4 \cosh{b(r - \kappa R)} \\
\end{aligned}
\end{equation}

so at $r = R$:

\begin{equation}
\begin{aligned}
    \frac{df}{dr} &= bC_3 \sinh{\phi(1 - \kappa)} + bC_4 \cosh{\phi(1 - \kappa)} \\
                  &= b \frac{\kappa \phi R}{D} \sinh{\phi(1 - \kappa)} + b \frac{R}{D} \cosh{\phi(1 - \kappa)} \\
                  &= \frac{\phi}{D} \left[ \kappa \phi \sinh{\phi(1 - \kappa)} + \cosh{\phi(1 - \kappa)} \right]\\
\end{aligned}
\end{equation}

Let $E = \kappa \phi \sinh{\phi(1 - \kappa)} + \cosh{\phi(1 - \kappa)} $:

\begin{equation}
    \frac{df}{dr} \bigg|_{r=R} = \frac{\phi E}{D}
\end{equation}

At $r = R$ we have:

\begin{equation}
    \frac{c_A}{c_{AR}} = \frac{f(R)}{R} = 1
\end{equation}

Therefore:

\begin{equation}
\begin{aligned}
    \frac{d c_A}{d r} \bigg|_{r=R} &= \frac{c_{AR}}{R^2} \left( \frac{R \phi E}{D} - R \right)
\end{aligned}
\end{equation}

So we can write:

\begin{equation}
\begin{aligned}
    W_{AR} &= - 4 \pi R^2 \mathcal{D}_A \frac{c_{AR}}{R} \left( \frac{\phi E}{D} - 1 \right) \\
           &= 4 \pi R \mathcal{D}_A c_{AR} \left( 1 - \frac{\phi E}{D} \right) \\
\end{aligned}
\end{equation}

Which can be expressed fully as:

\begin{equation}
\boxed{
     W_{AR} = 4 \pi R \mathcal{D}_A c_{AR} \left[ 1 - \phi \left( \frac{\kappa \phi \sinh{\phi(1 - \kappa)} + \cosh{\phi(1 - \kappa)}}{\kappa \phi \cosh{\phi(1 - \kappa)} + \sinh{\phi(1 - \kappa)}} \right)  \right]
}
\end{equation}
