\textbf{5--9, Problem Statement:}
Consider application of the conservation principles of Chapter 2 to control volume $A-B-C-D$ for the steady laminar boundary-layer problem sketched in Fig. 5P-9.

\begin{figure}[H]
    \centering
    \includegraphics[width=0.8\textwidth]{ME5130_HW03_Figure5P_9.png}
\end{figure}

\begin{enumerate}
    \item[(a)] Show that the mass flow rate across $B-C$ equals
    
    \begin{equation}
        \int_{0}^{y} (\rho U - \rho u) \, dy = \int_{x_A}^{x_D} \rho v(y) \, dx
    \end{equation}

    \item[(b)] Show that the drag force exerted on the fluid by the plate 
    
    \begin{equation}
        \int_{0}^{y}\frac{\rho UU}{g_c} \, dy - \left[ \int_{0}^{y}\frac{\rho uu}{g_c} \, dy + \int_{x_A}^{x_B} \frac{\rho v(y) U}{g_c} \, dx \right] - F_{\text{drag plate on fluid}} = 0
    \end{equation}

    \item[(c)] Use the results of parts a and b to show that, when $y \to \infty$ to avoid shear stresses that exist near the plate,
    
    \begin{equation}
        F_{\text{drag}} = \frac{\rho}{g_c} \int_{0}^{\infty} u( U - u ) \, dy
    \end{equation}

    And

    \begin{equation}
        \delta_2 = x \frac{\bar{C}_f}{2} = 0.66412 \left( \frac{\nu x}{U} \right)^{\frac{1}{2}}
    \end{equation}
\end{enumerate}

\textbf{5--9, Solution:}

\textbf{WE SHOULD LIST ASSUMPTIONS!!! HERE AND WHEN THEY APPLY THROUGHOUT EACH SOLUTION STEP!!!}

For a fixed box, we can write the mass balance as:

\begin{equation}
    \frac{\partial}{\partial t}\int_{CV}\rho\,dV=-\oint_{CS}\rho\,(\mathbf V\cdot\mathbf n)\,dS
\end{equation}

Since the probem is steady state, we can write:

\begin{equation}
    0=\oint_{CS}\rho\,(\mathbf V\cdot\mathbf n)\,dS
\end{equation}

We should note that on verticle faces, $dA = dy$ and on horizontal faces, $dA = dx$. For face $A-B$, $u = U$ and $\mathbf{n} = -i$. Therefore:

\begin{equation}
\begin{aligned}
    \mathbf{V} \cdot \mathbf{n} &= -u 
                                &= -U
\end{aligned}
\end{equation}

For face $B-C$, $\mathbf{n} = j$, therefore:

\begin{equation}
\begin{aligned}
    \mathbf{V} \cdot \mathbf{n} &= v
\end{aligned}
\end{equation}

For face $C-D$, $u = u(y)$ and $\mathbf{n} = i$. Therefore:

\begin{equation}
\begin{aligned}
    \mathbf{V} \cdot \mathbf{n} &= u
\end{aligned}
\end{equation}

and for face $D-A$, the plate is impermeable, so fluid cannot pass through it and thus $v = 0$. Nevertheless, we can now write our mass balance as:

\begin{equation}
\begin{aligned}
    0                           &= \int_0^y\rho\,(-U)\,dy+\int_{x_A}^{x_D}\rho\,v\,dx+\int_0^y\rho\,u\,dy+0 \\
    \int_{x_A}^{x_D}\rho\,v\,dx &= \int_0^y\rho U\,dy-\int_0^y\rho u\,dy \\ 
\end{aligned}
\end{equation}

and further as:

\begin{equation}
\boxed{
    \int_{x_A}^{x_D}\rho\,v\,dx = \int_0^y(\rho U-\rho u)\,dy
}
\end{equation}

We can now write the momentum balance as:

\begin{equation}
    \sum F_x=\frac{1}{g_c}\oint_{CS}\rho\,u\,(\mathbf V\cdot\mathbf n)\,dS
\end{equation}


For face $A-B$, the contribution is:

\begin{equation}
    \frac{1}{g_c}\int_0^y\rho\,U(-U)\,dy=-\int_0^y\frac{\rho UU}{g_c}\,dy
\end{equation}

For face $B-C$, the contribution is:

\begin{equation}
    \frac{1}{g_c}\int_{x_A}^{x_D}\rho\,U\,v\,dx=\int_{x_A}^{x_D}\frac{\rho\,v\,U}{g_c}\,dx
\end{equation}

For face $C-D$, the contribution is:

\begin{equation}
    \frac{1}{g_c}\int_0^y\rho\,u\,u\,dy=\int_0^y\frac{\rho uu}{g_c}\,dy
\end{equation}

and for face $D-A$, the contribution is $0$.

We can write the pressure force acting on face $A-B$ as:

\begin{equation}
    F_{AB}=+py
\end{equation}

The pressure force acting on face $C-D$ as:

\begin{equation}
    F_{CD}=-py
\end{equation}

and the pressure forces acting on faces $B-C$ and $A-D$ as:

\begin{equation}
    F_{BC, \, AD} = 0
\end{equation}

Because there is no component along the $x$-direcetion. Next, the shear stress on face $B-C$ can be written as:

\begin{equation}
    \tau_{yx}=\mu\frac{\partial u}{\partial y}
\end{equation}

However, above the boundary layer, $u = U$, therefore the horizontal velocity is constant, and:

\begin{equation}
    \frac{\partial u}{\partial y}=0
\end{equation}

Thus:

\begin{equation}
    \tau_{yx}=0
\end{equation}

and the shear force is zero. Next, because gravity does not have a horizontal component:

\begin{equation}
    F_{x, \, \text{gravity}} = 0
\end{equation}

Finally, we are left with the drag force exerted on the fluid:

\begin{equation}
    F_x = -F_{\text{drag}}
\end{equation}

Therefore:

\begin{equation}
\begin{aligned}
    \sum F_x &= +py-py+0+0+0-F_{\text{drag}} \\
             &= -F_{\text{drag}}
\end{aligned}
\end{equation}

and further:

\begin{equation}
\begin{aligned}
    -F_{\text{drag}} &= \frac{1}{g_c}\left[\int_0^y\rho uu\,dy+\int_{x_A}^{x_D}\rho vU\,dx-\int_0^y\rho UU\,dy\right] \\
                     &=\int_0^y\frac{\rho uu}{g_c}\,dy+\int_{x_A}^{x_D}\frac{\rho vU}{g_c}\, dx-\int_0^y\frac{\rho UU}{g_c}\,dy\\
    0                &=\int_0^y\frac{\rho uu}{g_c}\,dy+\int_{x_A}^{x_D}\frac{\rho vU}{g_c}\,dx-\int_0^y\frac{\rho UU}{g_c}\,dy+F_{\text{drag}} \\
                     &= -\int_0^y\frac{\rho uu}{g_c}\,dy-\int_{x_A}^{x_D}\frac{\rho vU}{g_c}\,dx+\int_0^y\frac{\rho UU}{g_c}\,dy-F_{\text{drag}} \\
                     &= \int_0^y\frac{\rho UU}{g_c}\,dy-\int_0^y\frac{\rho uu}{g_c}\,dy-\int_{x_A}^{x_D}\frac{\rho vU}{g_c}\,dx-F_{\text{drag}}
\end{aligned}
\end{equation}

So finally:

\begin{equation}
\boxed{
    \int_0^y\frac{\rho UU}{g_c}\,dy-\left[\int_0^y\frac{\rho uu}{g_c}\,dy+\int_{x_A}^{x_D}\frac{\rho\,v\,U}{g_c}\,dx\right]-F_{\text{drag}} = 0
}
\end{equation}

We can now write:

\begin{equation}
\begin{aligned}
    F_{\text{drag}} &= \int_0^y\frac{\rho UU}{g_c}\,dy-\left[\int_0^y\frac{\rho uu}{g_c}\,dy+\int_{x_A}^{x_D}\frac{\rho vU}{g_c}\,dx\right] \\
                    &= \int_0^y\frac{\rho UU}{g_c}\,dy-\int_0^y\frac{\rho uu}{g_c}\,dy-\int_{x_A}^{x_D}\frac{\rho vU}{g_c}\,dx \\
                    &= \frac{\rho}{g_c}\int_0^y UU\,dy-\frac{\rho}{g_c}\int_0^y uu\,dy-\frac{\rho U}{g_c}\int_{x_A}^{x_D}v\,dx
\end{aligned}
\end{equation}

Now, from part a we have:

\begin{equation}
\begin{aligned}
    \rho\int_0^y(U-u)\,dy &= \rho\int_{x_A}^{x_D}v\,dx \\
    \int_0^y(U-u)\,dy     &= \int_{x_A}^{x_D}v\,dx
\end{aligned}
\end{equation}

Therefore we can write:

\begin{equation}
\begin{aligned}
    \frac{\rho U}{g_c}\int_{x_A}^{x_D}v\,dx &= \frac{\rho U}{g_c}\int_0^y(U-u)\,dy \\
                                            &= \frac{\rho}{g_c}\int_0^y(UU-Uu)\,dy \\
                                            &= \frac{\rho}{g_c}\int_0^y u(U-u)\,dy 
\end{aligned}
\end{equation}

So we now have:

\begin{equation}
\begin{aligned}
    F_{\text{drag}} &= \frac{\rho}{g_c}\int_0^y UU\,dy-\frac{\rho}{g_c}\int_0^y uu\,dy-\frac{\rho}{g_c}\int_0^y(UU-Uu)\,dy \\
                    &= \frac{\rho}{g_c}\int_0^y\Big[UU-uu-(UU-Uu)\Big]\,dy
\end{aligned}
\end{equation}

Therefore, as $y\to\infty$:

\begin{equation}
\boxed{
    F_{\text{drag}}=\frac{\rho}{g_c}\int_0^\infty u(U-u)\,dy
}
\end{equation}

We can observe that fluid above the boundary layer does not contribute to the drag force $(u = U)$. We can now write the definition of momentum thickness as:

\begin{equation}
    U^2\delta_2\equiv\int_0^\infty u(U-u)\,dy
\end{equation}

and therefore we can write:

\begin{equation}
    F_{\text{drag}}=\frac{\rho}{g_c}\,U^2\,\delta_2
\end{equation}

Now, we also know that:

\begin{equation}
    F_{\text{drag}}=\frac{1}{2}\,\frac{\rho}{g_c}\,U^2\,\bar C_f\,x
\end{equation}

So we can write:

\begin{equation}
\begin{aligned}
    \frac{\rho}{g_c}\,U^2\,\delta_2 &= \frac{1}{2}\,\frac{\rho}{g_c}\,U^2\,\bar C_f\,x \\
    \delta_2                        &= x\,\frac{\bar C_f}{2}
\end{aligned}
\end{equation}

We also know that:

\begin{equation}
    \frac{C_f}{2}=\left.\frac{d^2F}{d\eta^2}\right|_{\eta=0}Re_x^{-1/2}
\end{equation}

and from the Blasius solution:

\begin{equation}
    \left.\frac{d^2F}{d\eta^2}\right|_{\eta=0}=0.33206
\end{equation}

Thus:

\begin{equation}
\begin{aligned}
    \frac{C_f}{2} &= 0.33206\,Re_x^{-1/2} \\
    C_f         &= 0.66412\,Re_x^{-1/2}
\end{aligned}
\end{equation}

and since:

\begin{equation}
    \frac{\bar C_f}{2}=C_f
\end{equation}

We have:

\begin{equation}
    \frac{\bar C_f}{2}=0.66412\,Re_x^{-1/2}
\end{equation}

and further:

\begin{equation}
    \delta_2=x\,(0.66412)\,Re_x^{-1/2}
\end{equation}

Using $Re_x=Ux/\nu$, we can now write:

\begin{equation}
    \frac{1}{Re_x^{1/2}} = \frac{1}{(Ux/\nu)^{1/2}} = \left(\frac{\nu}{Ux}\right)^{1/2}
\end{equation}

So we have:

\begin{equation}
\begin{aligned}
    \delta_2 &= 0.66412\,x\left(\frac{\nu}{Ux}\right)^{1/2} \\
             &= 0.66412\,(x^2)^{1/2}\left(\frac{\nu}{Ux}\right)^{1/2} \\
             &= 0.66412\left(\frac{x^2\,\nu}{Ux}\right)^{1/2} \\
             &= 0.66412\left(\frac{x\,\nu}{U}\right)^{1/2}
\end{aligned}
\end{equation}

and therefore:

\begin{equation}
\boxed{
    \delta_2=x\,\frac{\bar C_f}{2}=0.66412\left(\frac{\nu x}{U}\right)^{1/2}
}
\end{equation}
